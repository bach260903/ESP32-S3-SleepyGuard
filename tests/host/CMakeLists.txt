cmake_minimum_required(VERSION 3.16)
project(bringup_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
add_library(bringup_logic STATIC ../../components/bringup_logic/bringup_logic.c)
target_include_directories(bringup_logic PUBLIC ../../components/bringup_logic/include)
add_executable(test_smoke test_smoke.c)
add_test(NAME smoke COMMAND test_smoke)
add_executable(test_startup_faults test_startup_faults.c)
target_include_directories(test_startup_faults PRIVATE ../../main)
target_link_libraries(test_startup_faults PRIVATE bringup_logic)
add_test(NAME startup_faults COMMAND test_startup_faults)
